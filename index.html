<script>
let isRunning1 = false;
let isRunning2 = false;
let breakerClosed = false;
let freq1 = 50.0, freq2 = 50.8;
let volt1 = 440, volt2 = 465;
let phaseAngle = 0.0;
let rotorAngle1 = 0.0;
let rotorAngle2 = 0.0;
const canvas = document.getElementById("synchroCanvas");
const ctx = canvas.getContext("2d");
function updateControls() {
freq1 = parseFloat(document.getElementById("freqSlider1").value);
volt1 = parseInt(document.getElementById("voltSlider1").value);
freq2 = parseFloat(document.getElementById("freqSlider2").value);
volt2 = parseInt(document.getElementById("voltSlider2").value);
document.getElementById("freqVal1").innerText = freq1.toFixed(1) + " Hz";
document.getElementById("voltVal1").innerText = volt1 + " V";
document.getElementById("freqVal2").innerText = freq2.toFixed(1) + " Hz";
document.getElementById("voltVal2").innerText = volt2 + " V";
}
function toggleMachine(num) {
if (num === 1) {
isRunning1 = !isRunning1;
const btn = document.getElementById("btn1");
const ind = document.getElementById("ind1");
if (isRunning1) {
btn.innerText = "Stop Machine 1";
ind.classList.add("on");
document.getElementById("messageLogBar").innerText = "Alternator 1 started and energized the main Busbar.";
} else {
btn.innerText = "Start Machine 1";
ind.classList.remove("on");
if (breakerClosed) {
document.getElementById("breakerSwitch").checked = false;
toggleBreaker(document.getElementById("breakerSwitch"));
}
isRunning2 = false;
document.getElementById("btn2").innerText = "Start Machine 2";
document.getElementById("ind2").classList.remove("on");
document.getElementById("messageLogBar").innerText = "Alternator 1 shut down.";
}
} else if (num === 2) {
if (!isRunning1) {
alert("WARNING: Start Alternator 1 (Busbar) first!");
return;
}
isRunning2 = !isRunning2;
const btn = document.getElementById("btn2");
const ind = document.getElementById("ind2");
if (isRunning2) {
btn.innerText = "Stop Machine 2";
ind.classList.add("on");
document.getElementById("messageLogBar").innerText = "Alternator 2 started. Adjust Freq & Volt to match Busbar.";
} else {
btn.innerText = "Start Machine 2";
ind.classList.remove("on");
if (breakerClosed) {
document.getElementById("breakerSwitch").checked = false;
toggleBreaker(document.getElementById("breakerSwitch"));
}
document.getElementById("messageLogBar").innerText = "Alternator 2 stopped.";
}
}
}
function toggleBreaker(checkbox) {
if (checkbox.checked) {
if (!isRunning2) {
checkbox.checked = false;
document.getElementById("messageLogBar").innerText = "ERROR: Alternator 2 is not running!";
return;
}
let freqDiff = Math.abs(freq1 - freq2);
let voltDiff = Math.abs(volt1 - volt2);
let phaseDiff = Math.abs(phaseAngle);
if (freqDiff < 0.3 && voltDiff < 20 && phaseDiff < 10) {
breakerClosed = true;
checkbox.checked = true;
document.getElementById("messageLogBar").innerText = `SUCCESS: Paralleled! Freq Diff: ${freqDiff.toFixed(1)} Hz, Volt Diff: ${voltDiff.toFixed(0)} V`;
} else {
breakerClosed = false;
checkbox.checked = false;
document.getElementById("messageLogBar").innerText = `DANGER: Out-of-phase fault! Freq Diff: ${freqDiff.toFixed(1)} Hz, Phase: ${phaseDiff.toFixed(1)}°`;
}
} else {
breakerClosed = false;
document.getElementById("messageLogBar").innerText = "Parallel breaker opened. Alt 2 isolated.";
}
}
setInterval(() => {
if (isRunning1) {
rotorAngle1 = (rotorAngle1 + freq1 * 0.8) % 360;
}
if (isRunning2) {
rotorAngle2 = (rotorAngle2 + freq2 * 0.8) % 360;
if (breakerClosed) {
freq2 = freq1;
phaseAngle = 0.0;
} else {
let slipFreq = freq2 - freq1;
if (Math.abs(slipFreq) < 0.01) {
phaseAngle = 0.0;
} else {
phaseAngle = (phaseAngle + slipFreq * 3.5);
if (phaseAngle > 180) phaseAngle -= 360;
if (phaseAngle < -180) phaseAngle += 360;
}
}
} else {
phaseAngle = 0.0;
}
if (isRunning1) {
document.getElementById("busVoltDisp").innerText = volt1 + " V";
document.getElementById("busFreqDisp").innerText = freq1.toFixed(1) + " Hz";
} else {
document.getElementById("busVoltDisp").innerText = "0 V";
document.getElementById("busFreqDisp").innerText = "0.0 Hz";
}
if (isRunning1 && isRunning2 && !breakerClosed) {
let freqDiff = freq2 - freq1;
let voltDiff = volt2 - volt1;
if (Math.abs(freqDiff) <= 0.2 && Math.abs(voltDiff) <= 15) {
document.getElementById("messageLogBar").innerText = "Parameters Matched! Wait for 12 o'clock & close switch.";
} else {
let fText = freqDiff > 0.2 ? "Decrease Freq" : (freqDiff < -0.2 ? "Increase Freq" : "Freq OK");
let vText = voltDiff > 10 ? "Decrease Volt" : (voltDiff < -10 ? "Increase Volt" : "Volt OK");
document.getElementById("messageLogBar").innerText = `F1:${freq1.toFixed(1)} | F2:${freq2.toFixed(1)} (${fText}) || V1:${volt1} | V2:${volt2} (${vText})`;
}
}
drawSynchroscopeScene();
}, 33);
function drawSynchroscopeScene() {
ctx.clearRect(0, 0, canvas.width, canvas.height);
let cx = canvas.width / 2;
let cy = 175;
ctx.beginPath();
ctx.arc(cx, 30, 18, 0, 2 * Math.PI);
ctx.fillStyle = breakerClosed ? "#00ff00" : "#2e5a2e";
ctx.fill();
ctx.lineWidth = 2;
ctx.strokeStyle = "#daa520";
ctx.stroke();
ctx.fillStyle = "#222";
ctx.font = "bold 12px Arial";
ctx.textAlign = "center";
ctx.fillText("Sync Success Lamp", cx, 60);
ctx.save();
ctx.translate(cx, cy);
ctx.beginPath();
ctx.arc(0, 0, 88, 0, 2 * Math.PI);
ctx.fillStyle = "#26262e";
ctx.fill();
ctx.beginPath();
ctx.arc(0, 0, 78, 0, 2 * Math.PI);
ctx.fillStyle = "#fcfcfc";
ctx.fill();
ctx.strokeStyle = "#666";
ctx.lineWidth = 2;
ctx.stroke();
for (let t = 0; t < 360; t += 30) {
let rad = (t * Math.PI) / 180;
let rIn = t % 90 === 0 ? 62 : 68;
let rOut = 74;
ctx.beginPath();
ctx.moveTo(rIn * Math.cos(rad), rIn * Math.sin(rad));
ctx.lineTo(rOut * Math.cos(rad), rOut * Math.sin(rad));
ctx.strokeStyle = "#222";
ctx.lineWidth = t % 90 === 0 ? 2 : 1;
ctx.stroke();
}
ctx.fillStyle = "#cc0000";
ctx.font = "bold 10px Arial";
ctx.fillText("SLOW", -50, 4);
ctx.fillText("FAST", 32, 4);
ctx.fillStyle = "#333";
ctx.font = "bold 8px Arial";
ctx.fillText("SYNCHROSCOPE", 0, 48);
ctx.beginPath();
ctx.moveTo(0, -70);
ctx.lineTo(-5, -58);
ctx.lineTo(5, -58);
ctx.fillStyle = "#28a745";
ctx.fill();
let mismatch = (Math.abs(volt1 - volt2) / 480) + (Math.abs(phaseAngle) / 180);
let lampColor = mismatch < 0.05 ? "#444" : `rgb(${Math.floor(255*Math.min(1,mismatch))}, ${Math.floor(200*Math.min(1,mismatch))}, 20)`;
ctx.beginPath();
ctx.arc(-105, 0, 14, 0, 2 * Math.PI);
ctx.fillStyle = lampColor;
ctx.fill();
ctx.stroke();
ctx.beginPath();
ctx.arc(105, 0, 14, 0, 2 * Math.PI);
ctx.fillStyle = lampColor;
ctx.fill();
ctx.stroke();
ctx.fillStyle = "#222";
ctx.font = "bold 10px Arial";
ctx.fillText("Lamp L", -105, 24);
ctx.fillText("Lamp R", 105, 24);
let needleRad = ((phaseAngle - 90) * Math.PI) / 180;
ctx.beginPath();
ctx.moveTo(0, 0);
ctx.lineTo(62 * Math.cos(needleRad), 62 * Math.sin(needleRad));
ctx.strokeStyle = "#222";
ctx.lineWidth = 3.5;
ctx.stroke();
ctx.beginPath();
ctx.arc(0, 0, 10, 0, 2 * Math.PI);
ctx.fillStyle = "#888";
ctx.fill();
ctx.stroke();
ctx.restore();
drawAlternatorUnit(100, 290, isRunning1, rotorAngle1, "Alternator 1");
drawAlternatorUnit(canvas.width - 100, 290, isRunning2, rotorAngle2, "Alternator 2");
}
function drawAlternatorUnit(x, y, isRunning, rotorAngle, label) {
ctx.beginPath();
ctx.arc(x, y, 65, 0, 2 * Math.PI);
ctx.fillStyle = "#2c5282";
ctx.fill();
